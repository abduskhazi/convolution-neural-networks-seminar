\section{Problem Description}

\begin{frame}[t]{What is our problem?}
\begin{itemize}
\item In this paper the authors are trying to estimate the function
$f:\mathbb \alpha^m \to \{Binding, NonBinding\}^m$
where $m$ is the length of any ligand binding protein and $\alpha \in $
\{Set of amino acids possible in a protein\}.

\item They use Deep Convolutional Neural Networks for this purpose.

\item This function is learnable because the binding sites are directly influenced by the tertiary protein structure which in turn depends on its primary structure.

\item Notice that $m$ is variable here!!!

\item Previous attempts to solve this problem have used
SVM, Deep Neural networks, Bayes classifier, 3D Template matching (albeit with high computational costs).
\end{itemize}
\end{frame}

\section{Proposed Solutions}

\begin{frame}[t]{Proposed Solutions}

\begin{itemize}
\item Neural Machine Translation (NMT).
Binding-Site prediction can be formulated as a specific case of Neural machine Translation (NMT).
Hence, any solution of NMT can be used for our problem.
\item An amino acid sequence (being variable in length) can be formulated as a Recurrent neural network (RNN) problem, we can use RNN solutions for our problem
\item There are disadvantages e.g. Computational time (among others)

\item Solution - Convolution Neural Networks.
\begin{itemize}

\item Temporal-CNNs can be parallelized, are 10x faster and give state of the art results.

\item In Temporal-CNN (DeepCSeqSite) architecture, both local and global dependencies can be captured by increasing the number of CNN layers. More on this in later slides!!
\end{itemize}
\end{itemize}
\end{frame}

\section{CNN As Solution}

\begin{frame}[t]{CNNs as a solution}
\begin{itemize}
\item Let $\alpha \in $ \{Ala, Arg, Asn ...\}. i.e Set of amino acids.

\item A protein of length $m$ is represented as ${\alpha}^m$

\item A classifier that classifies ${\alpha}^m$ to $\{Binding, NonBinding\}^m$ is trained.

\item During training, a mini-batch of $B$ proteins is taken together.  Each protein $\alpha^m$ in the batch is padded with $p_i$ dummy residues given by
$$
p_i =  \max(\{m_1, m_2 \cdots m_B\})- m_i
$$
to make the lengths of proteins in a single batch equal.

\item The padding is removed before loss calculation.

\item The lengths of the batches can differ from each other.
\end{itemize}
\end{frame}

\begin{frame}[t]{CNNs as a solution}
\begin{itemize}
\item Each amino-acid in a protein is mapped to $\mathbb{R}^{30}$. Thus a protein is mapped to a feature space $\mathbb{R}^{m \times 30}$

\item $\mathbb{R}^{30}$ are manually engineered features given by
\begin{itemize}
\item \textbf{Position-Specific Score Matrix (PSSM)} $\in \mathbb{R}^{20}$. 
\item \textbf{Relative solvent accessibility} $\in \mathbb{R}$.
\item \textbf{Secondary Structure} i.e $[\alpha$-helix $\beta$-strand coil$]^T$ $\in \mathbb{R}^3$.
\item \textbf{Dihedral angle} - $ \mathbf{A} \in \mathbb{R}^2$.
\item \textbf{Conservation rating score} during evolution.  $\in \mathbb{R}^2$. 
\item \textbf{Residue type} $\in \mathbb{R}$.
\item \textbf{Position embeddings} $\in \mathbb{R}$.
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[t]{CNNs as a solution - Architecture}
\textbf{Effective Context Scope / Input Field}

When we apply Kernel to both $\mathbf{I}$ and $\mathbf{H}$ with stride 1,\\

$Context(\mathbf{O}_0) = \{H_0, H_1, H_2\}$\\
$Input Field(\mathbf{H}_0) = \{I_0, I_1, I_2\}$\\
$Input Field(\mathbf{H}_1) = \{I_1, I_2, I_3\}$\\
$Input Field(\mathbf{H}_2) = \{I_2, I_3, I_4\}$\\
$Input Field(\mathbf{O}_0) = Input Field(\mathbf{H}_0) \cup 
                                                   Input Field(\mathbf{H}_1) \cup 
                                                   Input Field(\mathbf{H}_2)$\\
$Input Field(\mathbf{O}_0) = \{I_0, I_1, I_2, I_3, I_4\}$\\
$|Input Field(\mathbf{O}_0)| = 5 = 1 + 2(3 -1) = 1 + n(k-1)$


\begin{itemize}
\item Hence a single convolutional node in $n^{th}$ convolutional layer (depth) with kernel width $k$ and stride 1 has an input field $1+ n(k-1)$. 

\item The authors propose an increase in $n$ i.e the depth of CNN instead of $k$ to prevent huge growth in optimization parameters.
\end{itemize}
\end{frame}

\subsection{Basic Design}

\begin{frame}[t]{Basic Design - Standard DeepCSeq-Site}
Here we consider batch size $B = 1$ for simplicity.
Preprocessed protein $\textbf{P} \in \mathbb{R}^{m \times d}$ is input to the DeepCSeq network.
Here $d = 30$ and $m$ is the length of the protein in the batch.
\begin{itemize}

\item Let Kernel $\mathbf{K} \in \mathbb{R}^{k \times d}$ be the 1-D kernel in our convolutions.

\item We apply same convolutions using these kernels to maintain length $m$.
\ifx false
\item $\mathbf{0}^{\frac{k}{2} \times d}$ is padded at both ends of \textbf{P} to maintain length $m$. This padding is also done at each step throughout the network for the same reason.\fi

\item To capture local co-relation and covalent bonds between adjacent residues,
the following convolution is applied
$$
[\mathbf{K}_1(\textbf{P})\:\: 
\mathbf{K}_2(\textbf{P})\:\:
 \cdots \mathbf{K}_{2c}(\textbf{P})]^T  
\in \mathbb{R}^{m \times 1 \times 2c}
$$

\item $\mathbb{R}^{m \times 1 \times 2c}$ is input to $N$ Basic blocks which form the\\ encoder of the network.
\end{itemize}
\end{frame}

\subsection{Encoder Design}

\begin{frame}[t]{Basic Block - DeepCSeq-Site}
A basic block in the network is defined as
$$
Block : \mathbb{R}^{m \times 1 \times 2c} \to \mathbb{R}^{m \times 1 \times 2c}
$$

A series of $N$ blocks are applied in the following form
$$
Block_N(Block_{N-1}.... Block_2(Block_1(\mathbf{X})))); \mathbf{X} \in \mathbb{R}^{m \times 1 \times 2c}
$$

The following diagram shows the basic architecture of a basic block \cite{ResearchPaper} \\

\includegraphics[scale=0.5]{BasicBlock}

\end{frame}


\begin{frame}[t]{Basic Block - DeepCSeq-Site}
Solutions for gradient issues
\begin{itemize}
\item There is a skip connection from the input of the block to the output.
This solves the vanishing gradient issue. The block function can be defined as
$$
Block(\mathbf{X}) = \mathbf{X} + Convolution( GLU( Normalization(\mathbf{X}) ) )
$$

\item Normalization prevents unexpected gradient due to the skip connection summation.
\end{itemize}
\end{frame}

\subsection{Decoder Design}

\begin{frame}[t]{Decoder - Standard DeepCSeq-Site}
2 Fully connected (convolution) layers are defined for the decoder in the following way 
\begin{itemize}
\item \textbf{Layer 1} - $\mathbf{K}^{1 \times 1 \times 2c}$ kernels are applied to get the first convolution output $\mathbf{X}_{hidden} \in \mathbb{R}^{m \times 1 \times c}$.
$$
\mathbf{X}_{hidden} = [\mathbf{K}_1(\mathbf{X})\:\: \mathbf{K}_2(\mathbf{X}) \cdots  
\mathbf{K}_c(\mathbf{X})]^T;
\mathbf{X} \in \mathbb{R}^{m \times 1 \times 2c};
$$
Here drop-out regularization is used.

\item \textbf{Layer 2} - $\mathbf{K}^{1 \times 1 \times c}$ kernels are applied to get the final prediction $\mathbf{Y}$.
$$
\mathbf{Y} = [\mathbf{K}_1(\mathbf{X}_{hidden})\:\:
\mathbf{K}_2(\mathbf{X}_{hidden})]^T;
\mathbf{Y} \in \mathbb{R}^{m \times 1 \times 2};
$$
$\mathbf{Y}$ is passed through a soft-max layer to get the values\\
in the range $[0,1]$. 
\end{itemize}
\end{frame}

\begin{frame}[t]{Loss function - Standard DeepCSeq-Site}
The loss function used is the negative cross entropy (for 2-way softmax classifier) defined for a single amino-acid $i$ in the sequence as 
$$
J_i(\theta) = - \mathbf{Y}_i \cdot log(\hat{\mathbf{Y}}_i);
\mathbf{Y}_i , \hat{\mathbf{Y}}_i \in \mathbb{R}^2
$$
$$
y_i \in \{0,1\};
\hat{y_i} \in [0,1];
\sum \mathbf{Y}_i = \sum \hat{\mathbf{Y}}_i = 1
$$
The final loss function for a protein-ligand prediction is given by
$$
J(\theta) = \sum_{i}^{m} J_i(\theta) + \gamma * L2Norm(\theta)
$$
After training is complete the classification threshold of softmax is set to a value $< 0.5$ because of non binding skew in data.
\end{frame}

\begin{frame}[t]{Decoder - Enhanced DeepCSeq-Site}
The probability of being a binding amino acid depends on wether its neighbours are binding or not. So to facilitiate this,  the following enhanced decoder is proposed by the authors

\begin{itemize}
\item Additional "Context prediction" is given as input to the enhanced decoder.

\item Predictions $\mathbf{P}_{left}$, $\mathbf{P}_{right}$ and the encoder output are all
added to form enhanced encoder's output.

\item \textbf{Teacher forcing at a higher abstraction}.
During training, we can directly use labels as the input to context prediction usage process.
During testing/future-use, the output of the standard decoder is given as the input to the enhanced decoder.

\item Then the decoder used is exactly the same as the standard decoder.
\end{itemize}
\end{frame}

\section{Optimisation}

\begin{frame}[t]{Optimisation}
There are 3 main hyperparameters to optimize - number of basic blocks $N$
, kernel width $k$ and the number of channels $c$.
\begin{itemize}
\item Optimization of $k$,$n$ is performed independently.
\item The authors fix $N = 10$ while optimising $k$ and $k = 3$ while optimising $N$.
\item As per empirical evidence, $N=10$, $k=5$ and $c=256$ gives best results.
\item Also $\gamma = 0.2$ and drop out ratio 0.5 is used during training.
\item The optimization is done through stochastic gradient descent
\item To prevent data skew, the mini-batch selection should be such that the proportion
of binding to non-binding residues remains roughly the same to that of the whole training data.
\end{itemize}
\end{frame}

\begin{frame}[t]{Optimisation - Learning Rate}
The learning rate consists of 3 stages. Each stage has exponential decay.
\begin{itemize}
\item $10^{-5}$ is used for first stage. This is to deal with very steep gradients. (Optimization step $< 19999$)
\item $10^{-4}$ is used for second stage. This is to avoid falling into local minima that predicts all residues as negative.
There is a huge skew towards negative (non-binding) residues in our data set.
(Optimization step $\in [20000 \to 99999]$)
\item $2 * 10^{-4}$ for the final stage till the end of the optimisation. (Optimization step $>99999$)
\end{itemize}
\end{frame}

\section{Evaluation}

\begin{frame}[t]{Evaluation}
Let $TP$, $TN$, $FP$ and $FN$ be the number of True positive, True
negative, False positive and False negative values of our prediction respectively.
3 metrics are used for evaluation. 
\begin{itemize}
\item Mathew's correlation coefficient,  $MCC$
$$
MCC = \frac{TP * TN - FP * FN}{\sqrt{(TP+FP)(TP+FN)(TN+FP)(TN+FN)}}
$$
\item Precision
$$
precision = \frac{TP}{TP + FP}
$$
\item Recall
$$
recall = \frac{TP}{TP + FN}
$$
\end{itemize}
\end{frame}


\section{Results}

\begin{frame}[t]{Results}
\begin{itemize}
\item The results were compared with 3D structure based models as they are the models
with best results - TM-SI, COF and COA
\item Standard DeepCSeq-Site is better than these by 0.05 in MCC score and 15\% in precision
\item There is bad Recall score because of non-binding bias in the data sets.
\item Enhanced DCSI improves MCC by $\approx 0.02$.  This is because of inclusion of 
more stronger context prediction input to the decoder
\item Data augmentation was not found to be suitable for the model due to high computational costs and low effect.
\end{itemize}
\end{frame}

\section{Discussion and Future Work}

\begin{frame}[t]{Discussion and conclusion}
\begin{itemize}
\item Traditional SVM and ANN do not utilize data effectively.  They have no feature learning, 
low generalization and representational capacity.
\item Deep CNN are better models. They  self learn hierarchical features and learn more abstract features at greater depths. Due to availability of massive sequencing data, 
generalization of the deep networks is possible.
\item They have advantage of capturing long distance relationships at higher depths without depending on a particular input field window.
\item Non binding data skew ($\approx 96\%$)is better handled in DCS-SI as compared to SVM and other base-line models.\\
For example - generalization and convergence of \\
support vectors learned by SVM is difficult.
\end{itemize}
\end{frame}

\begin{frame}[t]{Future Work}
\begin{itemize}
\item Usage of attention mechanisms to capture long distance coorelations between
amino-acids.
\item Inclusion of more 3D structure data in the feature map.
\item Usage of Generative adverserial networks to tackle quantitative disparity between
3D structure data and sequence data.
\end{itemize}
\end{frame}

\section{Further Information/Details for Q and A...}


\begin{frame}[t]{Biological Background}
Protein formation process. \cite{DNAtoProtein}

\begin{itemize}
\item All proteins required by an organism are encoded in the DNA.

\item First, RNA polymerase transcribes the part of our DNA which contains protein encoding to messenger RNA a.k.a mRNA.

\item  mRNA is post-processed to get an RNA. This RNA goes outside the nucleus of the cell. RNA $\in \{A, T, C, G\}^n$ where $n \in \mathbb{I}^+$

\item  Ribosomes in the cell cytoplasm then translate RNA to an amino acide sequence. This sequence is called primary structure of the protein. The mapping can be written as - $Ribosome:\mathbb \{A, T, C, G\}^n \to \{R, N,D, C ...\}^m$\\
where $ m \approx \frac{n}{3}$ and $m \in \mathbb{I}^+$.  

\item This amino acid sequence folds into a 3D shape \\
depending on its sequence. This structure is called the \\
tertiary protein structure.
\end{itemize}
\end{frame}

\begin{frame}[t]{Binding-Residue Prediction vs NMT}

\begin{itemize}
\item Neural Machine Translation (NMT) can be defined as a function
$g:\mathbb {\alpha}^n \to {\beta}^m$ where $\alpha,\beta$ are alphabets of 2 distinct languages and $n,m \in \mathbb{I}^+$

\item Binding-Site prediction is a specific case of NMT, where $m = n$, $\alpha$ is one of  20 amino acids and $\beta$ is $\{Binding, NonBinding\}$.
 
\item The local and long distance coorelation between different ${\alpha}^n$ and ${\beta}^m$ exist in both NMT and BR prediction.

\item Hence, any solution of NMT can be used for our problem.

\item Fun fact - Ribosome is also "Translating" from DNA alphabet to Amino acid alphabet !!!
\end{itemize}
\end{frame}

\begin{frame}[t]{Previous solutions}
\begin{itemize}
\item Previous methods used sliding window concept to capture residue dependencies.

\item But, nearby residues in the tertiary structure may be located far away in its primary structures.
\item Just increasing window size is infeasible due to memory and computation costs.

\item The authors create a CNN architecture (named DeepCSeqSite) and use it as a universal function approximator.

\item The authors do not solve ligand specific prediction tasks.

\item They classify each amino-acid in the sequence as binding or non-binding.
\end{itemize}
\end{frame}

\begin{frame}[t]{RNNs as a solution}

\begin{itemize}
\item As our amino acid sequence is variable in length, we could use RNNs to capture long term dependencies in the sequence.

\item There are some disadvantages
\begin{itemize}

\item The first amino acid is processed $n$ times, where n is the length of the protein. And the last amino acid is processed only 1 time.

\item RNNs are sequential in principle.
Parallelizing the computation is not possible.
\end{itemize}
\end{itemize}

Solution - Convolution Neural Networks.
\begin{itemize}

\item Temporal-CNNs can be parallelized, are 10x faster and give state of the art results.

\item In Temporal-CNN (DeepCSeqSite) architecture, both local and global dependencies can be captured by increasing the number of CNN layers. 
\end{itemize}
\end{frame}

\begin{frame}[t]{Input Feature Map Calculation}
\begin{itemize}
\item $\mathbb{R}^{30}$ are manually engineered features given by
\begin{itemize}
\item \textbf{Position-Specific Score Matrix (PSSM)} $\in \mathbb{R}^{20}$. This is the probilility of mutation to other amino acids at each position.
Since this  mutation is not position independent,  it is calculated for the whole protein i.e $\in \mathbb{R}^{m \times 20}$.  Normalized values are calculated by
$$
Norm(\mathbf{x}) = \frac{1}{1+2^{-\mathbf{x}}}; \mathbf{x}, \in \mathbb{R}^{m \times 20}
$$

\item \textbf{Relative solvent accessibility} $\in \mathbb{R}$.
Buried $(< 0.25)$ or exposed $(> 0.25)$ in the tertiary-structure.

\item \textbf{Secondary Structure} i.e $[\alpha$-helix $\beta$-strand coil$]^T$ $\in \mathbb{R}^3$.\\
$\alpha$-helix,$\beta$-strand, coil $\in [0,1]$ indicating probabilities of \\
existence.

\item \textbf{Dihedral angle} - $ \mathbf{A} \in \mathbb{R}^2$ where $Norm(\mathbf{A}) = \frac{\mathbf{A}}{360}$.
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[t]{Input Feature Map Calculation}
\begin{itemize}
\item \textbf{Conservation rating score} during evolution. 
$[$Relative-entropy Jensen-Shannon-divergence$]^T \in \mathbb{R}^2$. 

\item \textbf{Residue type} $\in \mathbb{R}$ given by
$$
r = \frac{type}{21}; type \in [0,20]; type \in \mathbb{I}^+
$$
Here $0$ means dummy residue, $[1-20]$ represents other residues.

\item \textbf{Position embeddings} $\in \mathbb{R}$ given by 
$$
PE_i = \frac{i}{L}
$$
for residue at $i^{th}$ position where $L$ is the length of the\\
protein.
\end{itemize}
\end{frame}

\begin{frame}[t]{Basic Block - DeepCSeq-Site}
\begin{itemize}
\item  First input $\mathbf{X} \in \mathbb{R}^{m \times 1 \times 2c}$ is layer normalized
to get $\mathbf{X}_{norm}$
\ifx false
$$
\mathbf{X}_i = \frac{\mathbf{X}_i - mean(\mathbf{X}_i)}{StdDeviation(\mathbf{X}_i)};  \forall \mathbf{X}_i \in  \mathbb{R}^{m \times 1}; i \in [1,2c], \mathbb{I}^+
$$
\fi

\item Next, $\mathbf{X}_{norm} = [\mathbf{A}: \mathbf{B}]$ where 
$\mathbf{A},\mathbf{B} \in \mathbb{R}^{m \times 1 \times c}$ is passed through a Gated Linear Unit which performs the following operation to get $\mathbf{X}_{gated}$
$$
GLU: \mathbb{R}^{m \times 1 \times 2c} \to \mathbb{R}^{m \times 1 \times c}
$$
$$
\mathbf{X}_{gated} = GLU([\mathbf{A}: \mathbf{B}]) = \mathbf{A} \odot sigmoid(\mathbf{B})
$$

\item $\mathbf{K} \in \mathbb{R}^{k \times 1 \times c}$ 1-D kernels are applied to $\mathbf{X}_{gated}$ to get\\ $\mathbf{X}_{output} \in \mathbb{R}^{m \times 1 \times 2c}$\\
$
\mathbf{X}_{output}  = [\mathbf{K}_1(\mathbf{X}_{gated}) \cdots \mathbf{K}_{2c}(\mathbf{X}_{gated})]^T
$
\end{itemize}
\end{frame}

\begin{frame} [t]{Context Prediction}
Using Left Context prediction
\begin{itemize}
\item Let $\mathbf{p}$ be the context/prediction of our protein.

\item $|\mathbf{p}| = m$ where $m$ is the length of the protein.

\item The last amino-acid is not at the left of any other amino-acid.
So it is not necessary for calculation.

\item Hence $\mathbf{p}$ is transformed by pre-pending $0$s to form\\
 $\mathbf{p^\prime} = [0\:\:0 \cdots $k zeros$] : [p_1\:\: p_2\:\: \cdots p_{m-1}]$

\item Kernels of size $\mathbf{K} \in \mathbb{R}^{k \times 1}$ are applied to $\mathbf{p^\prime} $ to get $\mathbf{P}_{left} \in \mathbb{R}^{m \times 1 \times 2c}$
$$
\mathbf{P}_{left} = [\mathbf{K}_1(\mathbf{p^\prime})\:\: \mathbf{K}_2(\mathbf{p^\prime} )\:\: \cdots \mathbf{K}_{2c}(\mathbf{p^\prime})]^T
$$
\item Usage of Right Context prediction is just a mirror image of the Left.
\end{itemize}
\end{frame}
